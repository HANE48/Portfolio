-- 이름이 Michael 이고 직종이 MK_MAN인 사원의 급여를 초과하는 사원들의 정보를 사번, 이름, 직종, 급여 순으로 출력

SELECT employee_id, first_name, job_id, salary
FROM EMPLOYEES
WHERE salary > ( 
  SELECT salary
  FROM EMPLOYEES
  WHERE first_name = 'Michael' AND job_id = 'MK_MAN' 
);

-- 사번이 111번인 사원의 직종과 같고, 159번 사원의 급여를 초과하는 사원들의 정보를 
사번, 이름, 직종, 입사일, 급여, 순으로 출력
SELECT employee_id, first_name, job_id, hire_date, salary

  FROM EMPLOYEES

) AND salary > (
  SELECT salary
  FROM EMPLOYEES

-- 회사에서 가장 월급을 많이 받는 사원의 사번, 이름, 월급을 출력
SELECT employee_id, first_name, salary
FROM EMPLOYEES
WHERE salary = (
  SELECT MAX(salary)
  FROM EMPLOYEES
);

-- 월급의 평균이 5000이상인 부서들을 찾고, 해당 부서의 인원수를 출력하되, 많은 순으로 정렬해서 부서번호, 인원수를 보여주기
SELECT department_id, count(*) cnt
FROM EMPLOYEES
WHERE department_id IS NOT NULL
GROUP BY department_id
HAVING AVG(salary) >= 5000
ORDER BY cnt DESC;

-- 유럽지역 사원들의 평균 급여보다 많이 받는 사원들의 이름, 급여, 지역을 출력
SELECT e.first_name, e.salary, r.region_name
FROM EMPLOYEES e, LOCATIONS l, DEPARTMENTS d, REGIONS r, COUNTRIES c
WHERE e.department_id = d.department_id
AND d.location_id = l.location_id
AND l.country_id = c.country_id
AND c.region_id = r.region_id
AND salary > (
  SELECT AVG(salary)
  FROM EMPLOYEES e, LOCATIONS l, DEPARTMENTS d, REGIONS r, COUNTRIES c
  WHERE e.department_id = d.department_id
  AND d.location_id = l.location_id
  AND l.country_id = c.country_id
  AND c.region_id = r.region_id
  AND r.region_name = 'Europe'
)

SELECT e.first_name, e.salary, r.region_name
FROM EMPLOYEES e
JOIN DEPARTMENTS d ON e.department_id = d.department_id
JOIN LOCATIONS l ON  d.location_id = l.location_id
JOIN COUNTRIES c ON l.country_id = c.country_id
JOIN REGIONS r ON c.region_id = r.region_id
WHERE e.salary > (
  SELECT AVG(salary)
  FROM EMPLOYEES e
  JOIN DEPARTMENTS d ON e.department_id = d.department_id
  JOIN LOCATIONS l ON  d.location_id = l.location_id
  JOIN COUNTRIES c ON l.country_id = c.country_id
  JOIN REGIONS r ON c.region_id = r.region_id 
  WHERE r.region_name = 'Europe'
)

-- 경력이 가장 오래된 사원의 사번, 이름, 급여, 입사일, 부서이름을 출력

-- 근무지가 시애틀인 사람들의 평균 급여보다 많이 받는 사원들의 사번, 이름, 급여를 조회
SELECT e.employee_id, e.first_name, e.salary
FROM EMPLOYEES 
WHERE salary > (
  SELECT AVG(salary)
  FROM EMPLOYEES e
  JOIN DEPARTMENTS D ON e.department_id = d.department_id
  JOIN LOCATIONS L ON d.location_id = l.location_id
  WHERE l.city = 'Seattle'
)

-- 시애틀근무자들 중에서 시애틀 평균 월급보다 많이 받는 사원들을 추려서 이름, 급여, 도시 순으로 출력
SELECT first_name, salary, l.city
FROM EMPLOYEES e
JOIN DEPARTMENTS d ON e.department_id = d.department_id
JOIN LOCATIONS l ON d.location_id = l.location_id
WHERE city = 'Seattle' AND salary > (
  SELECT AVG(salary)
  FROM EMPLOYEES e
  JOIN DEPARTMENTS d ON e.department_id = d.department_id
  JOIN LOCATIONS l ON d.location_id = l.location_id
  WHERE city = 'Seattle'
);
