5) SUM(): 총 합을 구하는 함수

-- 문자 관련 함수(그룹함수는 아님)  -- 
1) initcap(): 첫 글자를 대문자로 변환
2) lower(): 대문자를 소문자로 변환
3) upper(): 소문자를 대문자로 변환
4) lpad(), rpad(): 왼쪽 또는 오른쪽으로 원하는 byte만큼 자리를 확보
5) substr(): 특정 문장을 잘라내는 함수
6) replace(): 원하는 문자를 교체하는 함수
7) length(): 문자열의 길이를 가져옴
8) instr(): 특정 문자열의 위치를 찾아 반환하는 함수
9) ltrim(), rtrim(): 왼쪽 또는 오른쪽의 문자를 지우는 함수

SELECT initcap('good morning') FROM DUAL
SELECT INITCAP('good/morning') FROM DUAL 
SELECT LOWER('HOME COMMING') FROM DUAL
SELECT UPPER('hello world') FROM DUAL

-- 날짜함수
 - 현재 날짜를 가져오는 자료형: sysdate
SELECT sysdate FROM DUAL

1) months_between: a 와 b 사이에 개월 수
-- 입사일로부터 오늘까지 경과한 개월수
SELECT sysdate, hire_date, month_between(sysdate, hire_date) 
FROM EMPLOYEES;

2) add_months: 특정 날짜로부터 경과한 개월수에 따른 날짜
SELECT add_months(sysdate, 6) FROM DUAL

-- 입사 기간이 220개월 이상인 사원들의 이름, 입사일, 입사 후 현재까지 개월수를 입사 기간이 긴 사람 순으로 조회
(단, 입사 개월수는 소수점 이하로 절삭)
SELECT first_name, hire_date, trunc(months_between(sysdate, hire_date)) month
FROM EMPLOYEES
WHERE months_between(sysdate, hire_date) >= 220 
ORDER BY hire_date;

-- 120번 사원이 입사후 3년 6개월 만에 퇴사했다
사원의 사번, 이름, 입사일, 퇴사일을 출력
SELECT employee_id, first_name, hire_date, add_months(hire_date, 42)
FROM EMPLOYEES
WHERE employee_id = 120

-- 직종이 SA_REP인 사원들의 입사일로 부터 2년뒤 날짜를 사번, 이름, 입사일, 2년후 날짜 순으로 출력
SELECT employee_id, first_name, hire_date, add_months(hire_date, 24) after2
FROM EMPLOYEES
WHERE job_id = 'SA_REP'

-- to_char() --
날짜나 시간등의 특정 포맷을 지정할 수 있도록 하는 함수
1) YYYY 또는 YY : 년도
2) MM: 월
3) DD : 일
4) hh: 시
	4-1) hh24 : 24시간으로 표기
	4-2) hh12 : 12시간으로 표기(기본)
5) ss: 초
6) day: 요일
7) scc : 몇세기인지 나타냄
8) mi: 분
9) month : 월
	9-1) mon : 월을 3자리로 표기해줌


SELECT to_char(sysdate, 'yyyy-mm-dd day hh24:ss') day FROM DUAL

-- 모든 사원들의 이름과 입사년도를 '년-월-일 요일' 형태로 출력
SELECT first_name, to_char(hire_date, 'yyyy-mm-dd day') hire_date
FROM EMPLOYEES

-- 기타 함수 -- 
NVL() : NULL 값을 다른 데이터로 변경하는 함수

--보너스를 받지 않는 사원들의 commission을 0으로 수정
SELECT employee_id, commission_pct, nvl(commission_pct, 0)
FROM EMPLOYEES
WHERE commission_pct IS NULL;

--- RANK() OVER(): 순위를 매기는 함수
-- 부서번호,  사번, 이름, 월급을 월급을 많이 받는사람 순으로 순위를 매겨서 조회
SELECT department_id, employee_id, first_name, salary,
RANK() OVER(ORDER BY salary DESC)
FROM EMPLOYEES;

-- 입사일이 빠른 사람부터 순위를 매겨서 사번, 이름, 입사일, 순위 순으로 출력
SELECT employee_id, first_name, hire_date,
RANK() OVER(ORDER BY hire_date DESC) rank
FROM EMPLOYEES;

-- 사번, 이름, 월급을 조회, 월급이 낮은 순으로 순위를 매겨 출력
단 월급이 5000이상인 사원만 조회
SELECT employee_id, first_name, salary,
RANK() OVER(ORDER BY salary)
FROM EMPLOYEES
WHERE salary >= 5000;
