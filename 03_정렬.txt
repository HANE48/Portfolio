정렬( ORDER BY )
- 질의 결과의 반환되는 행(row)들을 정렬하고자 할 때 사용
- ORDER BY 절은 SELECT 절의 마지막에 기술
- 오름차순: ASC(생략 가능)
- 내림차순: DESC(생략 불가)

-- 급여가 15000 이상인 사원들의 모든 정보를 입사일이 빠른 순으로 정렬해서 조회
SELECT * FROM EMPLOYEES
WHERE salary >= 15000
ORDER BY hire_date ASC;

-- 월급을 많이 받는 사람들 부터 출력 월급이 같다면 입사일이 빠른 순으로, 사번, 이름, 급여, 입사일을 출력
SELECT employee_id, first_name, salary, hire_date
FROM EMPLOYEES
ORDER BY salary DESC, hire_date ASC

-- 입사일이 빠른 순으로 모든 정보를 출력하되 입사일이 같다면 이름, 내림차순으로 조회
SELECT * FROM EMPLOYEES
ORDER BY hire_date, first_name DESC;

-- 부서번호가 작은 것 부터 정렬 부서번호가 같으면 급여가 낮은 순으로 사번, 이름, 부서번호, 급여 순으로 출력
SELECT employee_id, first_name, department_id, salary
FROM EMPLOYEES
ORDER BY department_id, salary;

-- 부서번호 오름차순, 직종 오름차순, 급여 내림차순으로 사번, 이름, 부서 번호, 직종, 급여
SELECT employee_id, first_name, department_id, job_id, salary
FROM EMPLOYEES
ORDER BY department_id, job_id, salary DESC;

-- 부서번호가 80번이 아닌 나머지 부서의 사람들의 사번, 이름, 부서번호를 사번이 큰 사람부터 조회
SELECT employee_id, first_name, department_id
FROM EMPLOYEES
WHERE department_id != 80
ORDER BY employee_id DESC;

-- 직종이 ST_CLERK이 아닌 사원들의 사원들의 사번, 이름, 성, 급여, 직종을 직종내림차순으로 정렬
SELECT employee_id, first_name, last_name, salary, job_id
FROM EMPLOYEES
WHERE job_id NOT IN('ST_CLERK')
ORDER BY job_id DESC;

-- 월급이 4000미만인 사원들의 모든 정보를 사번이 큰 사람부터 정렬
SELECT * FROM EMPLOYEES
WHERE salary < 4000
ORDER BY employee_id DESC;

-- 급여가 5000이상 10000이하인 사원의 사번, 이름, 급여를 출력하고 급여가 낮은 순으로 정렬
SELECT employee_id, first_name, salary
FROM EMPLOYEES
WHERE salary >= 5000 AND salary <= 10000
ORDER BY salary ASC;

-- 부서번호가 30이거나 50인 사원들의 사번, 이름, 부서번호를 출력
   부서번호 오름차순, 이름 오름차순
SELECT employee_id, first_name, department_id
FROM EMPLOYEES
WHERE department_id = 30 OR department_id = 50 || WHERE department_id IN(30, 50)
ORDER BY department_id, frist_name;

-- 급여가 3000이하이거나 15000이상인 사원들의 사번, 이름, 급여를 급여가 높은순으로 조회
SELECT employee_id, first_name, salary 
FROM EMPLOYEE
WHERE salary NOT BETWEEN 3001 and 14999
ORDER BY salary DESC;

-- 직종이 SA_REP 고 급여가 7000이상인 사원들의 사번, 이름, 직종, 급여를 급여가 높은 순으로
SELECT employee_id, first_name, job_id, salary
FROM EMPLOYEES
WHERE job_id = 'SA_REP' 
AND salary >= 7000
ORDER BY salary DESC;

-- 2005년 이후 입사자들의 사번, 이름, 입사일을 일사일 최신순으로 정렬
SELECT employee_id, first_name, hire_date
FROM EMPLOYEES
WHERE hire_date >= '01/01/2005'
ORDER BY hire_date DESC;

-- 직종이 SA_MAN, IT_PROG, HR_REP인 사원들의 사번, 이름, 직종을 출력후 
직종->이름의 오름차순으로 정렬
SELECT employee_id, first_name, job_id
FROM EMPLOYEES
WHERE job_id IN('SA_MAN', 'IT_PROG', 'HR_REP')
ORDER BY job_id, first_name ;

-- 이름이 a로 시작하는 사원들의 사번, 이름을 출력 후 이름순으로 정렬
SELECT employee_id, first_name
FROM EMPLOYEES
WHERE first_name LIKE 'A%'
ORDER BY first_name;

-- 이름에 a가 포함된 사원들의 사번, 이름, 급여를 출력하는데 급여를 많이 받는 순으로 출력
SELECT employee_id, first_name, salary
FROM EMPLOYEES
WHERE first_name LIKE '%a%'
ORDER BY salary DESC;

-- 보너스를 받지 않는 사원들의 사번, 이름, 급여,  보너스를 출력하는데 급여를 많이 받는 순으로 정렬
SELECT employee_id, first_name, salary, commission_pct
FROM EMPLOYEES
WHERE commission_pct IS NULL
ORDER BY salary DESC;
